        un día no son independientes. `dedup=False` es la rama derogada."""

        grupos = _grupos_por_dia(df, diff)

        _, lo_t, hi_t = bf._ic_t_cluster(grupos)

        # vigilar esto no lo cazaba porque `mae_mejora_pct` es un cociente y el
